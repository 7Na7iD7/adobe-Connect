@echo off
echo ========================================
echo Finding files...
echo ========================================

for %%v in (*screenshare*.flv) do set video=%%v
for %%a in (*cameraVoip*.flv) do set audio=%%a

echo Video: %video%
echo Audio: %audio%

echo ========================================
echo Converting to MP4...
echo ========================================

ffmpeg -i "%video%" -i "%audio%" -c:v libx264 -preset ultrafast -c:a aac final_video.mp4

echo ========================================
echo Conversion completed!
echo File "final_video.mp4" is in this folder.
echo ========================================
pause
